# Pangolin v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a clean-slate Next.js app at `/Users/benjaminpoon/dev/pangolin-v2/web/` that turns Upwork listings into a prioritized micro-SaaS idea feed with AI extraction, deterministic viability scoring, and tagging.

**Architecture:** Listings captured by the existing Chrome extension land in an Inbox with `status='inbox'`. A "Process" button triggers batch AI extraction via Anthropic Haiku, which populates pain/solution fields and three scoring signal fields. A deterministic TypeScript function converts those signals into a 0–10 viability score. Ideas are then surfaced in a scored, tagged feed.

**Tech Stack:** Next.js 15 App Router, TypeScript, Tailwind CSS, PostgreSQL (`pg` Pool), Anthropic SDK (`claude-haiku-4-5-20251001`), Vitest (unit tests for pure functions).

```typescript
export interface ScoringInputs {
  budget_tier: string | null;
  pain_clarity: string | null;
  solution_saas_fit: string | null;
  recurrence_potential: string | null;
  tools_mentioned: string[] | null;
}

export interface ScoreBreakdown {
  budget: number;
  pain: number;
  fit: number;
  recurrence: number;
  tools: number;
  total: number;
}

const BUDGET_SCORES: Record<string, number> = { low: 1, mid: 2, high: 3 };
const PAIN_SCORES: Record<string, number> = { vague: 0, clear: 1, acute: 2 };
const FIT_SCORES: Record<string, number> = { unlikely: 0, possible: 1, strong: 2 };
const RECURRENCE_SCORES: Record<string, number> = { one_off: 0, recurring: 1, infrastructure: 2 };

export function computeViabilityBreakdown(idea: ScoringInputs): ScoreBreakdown {
  const budget     = BUDGET_SCORES[idea.budget_tier ?? ''] ?? 0;
  const pain       = PAIN_SCORES[idea.pain_clarity ?? ''] ?? 0;
  const fit        = FIT_SCORES[idea.solution_saas_fit ?? ''] ?? 0;
  const recurrence = RECURRENCE_SCORES[idea.recurrence_potential ?? ''] ?? 0;
  const tools      = Math.min((idea.tools_mentioned?.length ?? 0), 1);
  return { budget, pain, fit, recurrence, tools, total: budget + pain + fit + recurrence + tools };
}

export function computeViabilityScore(idea: ScoringInputs): number {
  return computeViabilityBreakdown(idea).total;
}
```

```typescript
export interface TaggingInputs {
  vertical: string | null;
  solution_pattern: string | null;
  recurrence_potential: string | null;
  solution_saas_fit: string | null;
  budget_tier: string | null;
  is_recurring_type_need: boolean | null;
  tools_mentioned: string[] | null;
}

export function deriveTags(idea: TaggingInputs): string[] {
  const tags: string[] = [];

  if (idea.vertical) tags.push(idea.vertical.toLowerCase());
  if (idea.solution_pattern) tags.push(idea.solution_pattern.toLowerCase());
  if (idea.recurrence_potential === 'infrastructure') tags.push('infrastructure');
  if (idea.recurrence_potential === 'recurring') tags.push('recurring-need');
  if (idea.solution_saas_fit === 'strong') tags.push('strong-fit');
  if (idea.budget_tier === 'high') tags.push('high-budget');
  if (idea.is_recurring_type_need) tags.push('recurring-type');
  (idea.tools_mentioned ?? [])
    .slice(0, 3)
    .forEach(t => tags.push(`tool:${t.toLowerCase()}`));

  return [...new Set(tags)];
}
```

```typescript
import type { ExtractionResult } from '@/types';

const client = new Anthropic();

const SYSTEM_PROMPT = `You are an expert at analyzing Upwork job listings to identify micro-SaaS product opportunities.

Extract the following and return ONLY valid JSON matching this exact schema — no markdown, no explanation:
{
  "vertical": string | null,
  "tools_mentioned": string[],
  "budget_tier": "low" | "mid" | "high" | null,
  "is_recurring_type_need": boolean | null,
  "pain_symptom": string | null,
  "pain_root_cause": string | null,
  "solution_specific": string | null,
  "solution_pattern": string | null,
  "saas_pitch": string | null,
  "pain_clarity": "vague" | "clear" | "acute" | null,
  "solution_saas_fit": "unlikely" | "possible" | "strong" | null,
  "recurrence_potential": "one_off" | "recurring" | "infrastructure" | null
}

Field guidance:
- vertical: industry/domain (e.g. "Marketing", "Finance", "Legal", "E-commerce")
- tools_mentioned: any software tools named in the listing
- budget_tier: low=under $500, mid=$500-$5000, high=over $5000
- pain_symptom: what the buyer describes experiencing (surface-level problem)
- pain_root_cause: the underlying cause of the pain
- solution_specific: what they specifically want built
- solution_pattern: category of solution (e.g. "Automation", "Integration", "Dashboard", "Analytics")
- saas_pitch: 3-5 sentences describing what SaaS product could solve this at scale for many buyers
- pain_clarity: vague=generic could apply to any business; clear=specific problem with identifiable use case; acute=urgent/expressed frustration/significant business impact
- solution_saas_fit: unlikely=highly custom/bespoke; possible=could be productized with effort; strong=clearly repeatable software problem with likely multiple buyers
- recurrence_potential: one_off=one-time project; recurring=ongoing subscription/retainer need; infrastructure=core business system with high retention if built`;

export async function extractIdea(
  title: string,
  description: string | null,
  skills: string[],
  budgetMin: number | null,
  budgetMax: number | null,
): Promise<{ result: ExtractionResult; rawText: string }> {
  const budgetContext =
    budgetMin != null || budgetMax != null
      ? `Budget: $${budgetMin ?? '?'} – $${budgetMax ?? '?'}`
      : 'Budget: not specified';

  const userContent = `Title: ${title}
${budgetContext}
Skills: ${skills.length > 0 ? skills.join(', ') : 'none listed'}
Description: ${description ?? '(no description provided)'}`;

  const message = await client.messages.create({
    model: 'claude-haiku-4-5-20251001',
    max_tokens: 1024,
    system: SYSTEM_PROMPT,
    messages: [{ role: 'user', content: userContent }],
  });

  const rawText =
    message.content[0].type === 'text' ? message.content[0].text : '';

  let parsed: Partial<ExtractionResult> = {};
  try {
    const match = rawText.match(/\{[\s\S]*\}/);
    if (match) parsed = JSON.parse(match[0]);
  } catch {
    parsed = {};
  }

  const validBudgetTiers = ['low', 'mid', 'high'] as const;
  const validPainClarity = ['vague', 'clear', 'acute'] as const;
  const validSaaSFit = ['unlikely', 'possible', 'strong'] as const;
  const validRecurrence = ['one_off', 'recurring', 'infrastructure'] as const;

  const result: ExtractionResult = {
    vertical:
      typeof parsed.vertical === 'string' ? parsed.vertical : null,
    tools_mentioned: Array.isArray(parsed.tools_mentioned)
      ? parsed.tools_mentioned.filter((t): t is string => typeof t === 'string')
      : [],
    budget_tier: validBudgetTiers.includes(parsed.budget_tier as typeof validBudgetTiers[number])
      ? (parsed.budget_tier as typeof validBudgetTiers[number])
      : null,
    is_recurring_type_need:
      typeof parsed.is_recurring_type_need === 'boolean'
        ? parsed.is_recurring_type_need
        : null,
    pain_symptom:
      typeof parsed.pain_symptom === 'string' ? parsed.pain_symptom : null,
    pain_root_cause:
      typeof parsed.pain_root_cause === 'string' ? parsed.pain_root_cause : null,
    solution_specific:
      typeof parsed.solution_specific === 'string' ? parsed.solution_specific : null,
    solution_pattern:
```

```typescript
export async function GET(req: NextRequest) {
  const { searchParams } = new URL(req.url);
  const status = searchParams.get('status') ?? 'saved';
  const tag = searchParams.get('tag');
  const limit = Math.min(parseInt(searchParams.get('limit') ?? '50'), 200);
  const offset = parseInt(searchParams.get('offset') ?? '0');

  const params: unknown[] = [status];

  let tagClause = '';
  if (tag) {
    params.push(tag);
    tagClause = ` AND EXISTS (SELECT 1 FROM idea_tags WHERE idea_id = i.id AND tag = $${params.length})`;
  }

  params.push(limit, offset);
  const sql = `
    SELECT i.*,
      COALESCE(array_agg(it.tag ORDER BY it.source, it.tag) FILTER (WHERE it.tag IS NOT NULL), '{}') AS tags
    FROM ideas i
    LEFT JOIN idea_tags it ON it.idea_id = i.id
    WHERE i.status = $1${tagClause}
    GROUP BY i.id
    ORDER BY i.viability_score DESC NULLS LAST, i.captured_at DESC
    LIMIT $${params.length - 1} OFFSET $${params.length}`;

  const ideas = await query(sql, params);

  const countParams: unknown[] = [status];
  let countSql = `SELECT count(*) FROM ideas i WHERE i.status = $1`;

    countParams.push(tag);
    countSql += ` AND EXISTS (SELECT 1 FROM idea_tags WHERE idea_id = i.id AND tag = $2)`;
  }
  const countRow = await queryOne<{ count: string }>(countSql, countParams);
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { query, queryOne } from '@/lib/db';

export async function GET(
  _req: NextRequest,
  { params }: { params: Promise<{ id: string }> },
) {
  const { id: idStr } = await params;
  const id = parseInt(idStr);
  if (isNaN(id)) return NextResponse.json({ error: 'Invalid id' }, { status: 400 });

  const idea = await queryOne(`SELECT * FROM ideas WHERE id = $1`, [id]);
  if (!idea) return NextResponse.json({ error: 'Not found' }, { status: 404 });

  const tags = await query<{ tag: string; source: string }>(
    `SELECT tag, source FROM idea_tags WHERE idea_id = $1 ORDER BY source, tag`,
    [id],
  );

  return NextResponse.json({ ...idea, tags });
}

export async function PATCH(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> },
) {
  const { id: idStr } = await params;
  const id = parseInt(idStr);
  if (isNaN(id)) return NextResponse.json({ error: 'Invalid id' }, { status: 400 });

  const body: { status?: string; addTag?: string; removeTag?: string } = await req.json();

  if (body.status) {
    if (!['inbox', 'saved', 'dismissed'].includes(body.status)) {
      return NextResponse.json({ error: 'Invalid status' }, { status: 400 });
    }
    await queryOne(`UPDATE ideas SET status=$1 WHERE id=$2`, [body.status, id]);
  }

  if (body.addTag) {
    await queryOne(
      `INSERT INTO idea_tags (idea_id, tag, source) VALUES ($1,$2,'manual') ON CONFLICT DO NOTHING`,
      [id, body.addTag.toLowerCase()],
    );
  }

  if (body.removeTag) {
    await queryOne(
      `DELETE FROM idea_tags WHERE idea_id=$1 AND tag=$2 AND source='manual'`,
      [id, body.removeTag],
    );
  }

  return NextResponse.json({ ok: true });
}
```

```tsx
    return <div className="p-8 text-gray-400 text-sm">Loading...</div>;
  }

  return (
    <div className="max-w-4xl mx-auto p-6">
      <div className="flex items-center justify-between mb-6">
        <div>
          <h1 className="text-2xl font-bold text-gray-900">Inbox</h1>
          <p className="text-sm text-gray-400 mt-0.5">
            {ideas.length} captures · {unprocessedCount} unprocessed
          </p>
        </div>
        {unprocessedCount > 0 && (
          <button
            onClick={processInbox}
            disabled={processing}
            className="bg-indigo-600 text-white px-4 py-2 rounded-lg text-sm font-medium hover:bg-indigo-700 disabled:opacity-50 transition-colors"
          >
            {processing ? 'Processing...' : `Process ${unprocessedCount}`}
          </button>
        )}
      </div>

      {processResult && (
        <div className="mb-4 p-3 bg-green-50 border border-green-200 rounded-lg text-sm text-green-700">
          Done — {processResult.succeeded} succeeded, {processResult.failed} failed.
          {processResult.failed > 0 && ' Check ai_error field for details.'}
        </div>
      )}

      {ideas.length === 0 && (
        <div className="text-center py-20 text-gray-400">
          <p className="text-lg font-medium">Inbox is empty</p>
          <p className="text-sm mt-1">Capture listings from Upwork using the Chrome extension.</p>
        </div>
      )}

      <div className="space-y-2">
        {ideas.map(idea => (
          <div
            key={idea.id}
            className="bg-white border border-gray-200 rounded-lg px-4 py-3 flex items-center gap-4"
          >
            <div className="flex-1 min-w-0">
              <p className="text-sm font-medium text-gray-900 truncate">{idea.title}</p>
              <div className="flex items-center gap-2 mt-1">
                <span className="text-xs text-gray-400">
                  {new Date(idea.captured_at).toLocaleDateString()}
                </span>
                {idea.source === 'single' && (
                  <span className="text-xs bg-purple-100 text-purple-700 px-1.5 py-0.5 rounded">
                    single
                  </span>
                )}
                {idea.budget_tier && (
                  <span className="text-xs bg-gray-100 text-gray-500 px-1.5 py-0.5 rounded">
                    {idea.budget_tier}
                  </span>
                )}
                {!idea.ai_processed_at && (
                  <span className="text-xs bg-yellow-100 text-yellow-700 px-1.5 py-0.5 rounded">
                    unprocessed
                  </span>
                )}
                {idea.ai_error && (
                  <span className="text-xs bg-red-100 text-red-600 px-1.5 py-0.5 rounded">
                    error
                  </span>
                )}
              </div>
            </div>
            <div className="flex gap-3 shrink-0">
              <button
                onClick={() => updateStatus(idea.id, 'saved')}
                className="text-xs font-medium text-indigo-600 hover:text-indigo-800"
              >
                Save
              </button>
              <button
                onClick={() => updateStatus(idea.id, 'dismissed')}
                className="text-xs text-gray-400 hover:text-gray-600"
              >
                Dismiss
              </button>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
```

```tsx
import { useState, useEffect, useCallback } from 'react';
import Link from 'next/link';

function ScoreBadge({ score }: { score: number | null }) {
  if (score == null) return null;
  const color =
    score >= 7
      ? 'bg-green-100 text-green-800'
      : score >= 4
      ? 'bg-yellow-100 text-yellow-800'
      : 'bg-red-100 text-red-800';
  return (
    <span className={`inline-flex items-center px-2 py-0.5 rounded text-xs font-bold ${color}`}>

  );

export default function IdeasPage() {
  const [ideas, setIdeas] = useState<IdeaListItem[]>([]);
  const [loading, setLoading] = useState(true);
  const [tagFilter, setTagFilter] = useState('');

  const load = useCallback(async (tag?: string) => {
    setLoading(true);
    const qs = tag ? `&tag=${encodeURIComponent(tag)}` : '';
    const res = await fetch(`/api/ideas?status=saved&limit=100${qs}`);
    const data = await res.json();

            className="text-sm text-indigo-600 hover:text-indigo-800"

            {tagFilter} ×
          </button>
```

```tsx
'use client';
import { useState, useEffect, useCallback } from 'react';
import { useRouter, useParams } from 'next/navigation';
import Link from 'next/link';
import type { Idea, IdeaTag } from '@/types';

interface IdeaDetail extends Idea {
  tags: IdeaTag[];
}

function ScoreBar({ score }: { score: number | null }) {
  if (score == null) return null;
  const pct = (score / 10) * 100;
  const barColor =
    score >= 7 ? 'bg-green-500' : score >= 4 ? 'bg-yellow-400' : 'bg-red-400';
  return (
    <div>
      <div className="flex justify-between items-center mb-1">
        <span className="text-sm font-medium text-gray-600">Viability Score</span>
        <span className="text-sm font-bold text-gray-900">{score} / 10</span>
      </div>
      <div className="h-2 bg-gray-100 rounded-full overflow-hidden">
        <div
          className={`h-full ${barColor} rounded-full transition-all`}
          style={{ width: `${pct}%` }}
        />
      </div>
    </div>
  );
}

function Field({ label, value, color }: { label: string; value: string | null; color: string }) {
  if (!value) return null;
  return (
    <div className={`rounded-lg p-4 ${color}`}>
      <p className="text-xs font-semibold uppercase tracking-wide mb-1 opacity-60">{label}</p>
      <p className="text-sm text-gray-800">{value}</p>
    </div>
  );
}

export default function IdeaDetailPage() {
  const { id } = useParams<{ id: string }>();
  const router = useRouter();
  const [idea, setIdea] = useState<IdeaDetail | null>(null);
  const [loading, setLoading] = useState(true);
  const [newTag, setNewTag] = useState('');
  const [saving, setSaving] = useState(false);

  const load = useCallback(async () => {
    const res = await fetch(`/api/ideas/${id}`);
    if (!res.ok) { router.push('/ideas'); return; }
    setIdea(await res.json());
    setLoading(false);
  }, [id, router]);

  useEffect(() => { load(); }, [load]);

  async function updateStatus(status: string) {
    setSaving(true);
    await fetch(`/api/ideas/${id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ status }),
    });
    setSaving(false);
    router.push(status === 'saved' ? '/ideas' : '/inbox');
  }

  async function addTag(e: React.FormEvent) {
    e.preventDefault();
    const tag = newTag.trim().toLowerCase();
    if (!tag) return;
    await fetch(`/api/ideas/${id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ addTag: tag }),
    });
    setNewTag('');
    load();
  }

  async function removeTag(tag: string) {
    await fetch(`/api/ideas/${id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ removeTag: tag }),
    });
    load();
  }

  if (loading) return <div className="p-8 text-gray-400 text-sm">Loading...</div>;
  if (!idea) return null;

  return (
    <div className="max-w-3xl mx-auto p-6 space-y-4">
      <Link href="/ideas" className="text-sm text-indigo-600 hover:text-indigo-800 inline-block mb-2">
        ← Ideas
      </Link>

      {/* Main card */}
      <div className="bg-white border border-gray-200 rounded-xl p-6 space-y-5">
        <h1 className="text-xl font-bold text-gray-900">{idea.title}</h1>
        <ScoreBar score={idea.viability_score} />

        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
          <Field label="Pain Symptom" value={idea.pain_symptom} color="bg-red-50" />
          <Field label="Root Cause" value={idea.pain_root_cause} color="bg-orange-50" />
          <Field label="Solution" value={idea.solution_specific} color="bg-blue-50" />
          <Field label="Pattern" value={idea.solution_pattern} color="bg-indigo-50" />
        </div>

        {idea.saas_pitch && (
          <div className="bg-green-50 rounded-lg p-4">
            <p className="text-xs font-semibold uppercase tracking-wide mb-1 text-green-700 opacity-80">
              SaaS Opportunity
            </p>
            <p className="text-sm text-gray-800">{idea.saas_pitch}</p>
          </div>
        )}

        {/* Score signals */}
        <div className="border-t border-gray-100 pt-4">
          <p className="text-xs font-semibold text-gray-400 uppercase tracking-wide mb-2">Score signals</p>
          <div className="flex flex-wrap gap-2 text-xs">
            {[
              { label: 'Budget', value: idea.budget_tier },
              { label: 'Pain', value: idea.pain_clarity },
              { label: 'Fit', value: idea.solution_saas_fit },
              { label: 'Recurrence', value: idea.recurrence_potential },
              { label: 'Tools', value: (idea.tools_mentioned?.length ?? 0) > 0 ? `${idea.tools_mentioned?.length} mentioned` : null },
            ].map(({ label, value }) =>
              value ? (
                <span key={label} className="bg-gray-100 text-gray-600 px-2 py-1 rounded">
                  {label}: {value}
                </span>
              ) : null,
            )}
          </div>
        </div>
      </div>

      {/* Tags card */}
      <div className="bg-white border border-gray-200 rounded-xl p-5">
        <p className="text-sm font-semibold text-gray-700 mb-3">Tags</p>
        <div className="flex flex-wrap gap-2 mb-3">
          {idea.tags.map(({ tag, source }) => (
            <span
              key={tag}
              className="inline-flex items-center gap-1 text-xs px-2.5 py-1 rounded-full bg-gray-100 text-gray-700"
            >
              {tag}
              {source === 'manual' && (
                <button
                  onClick={() => removeTag(tag)}
                  className="text-gray-400 hover:text-red-500 leading-none ml-0.5"
                >
                  ×
                </button>
              )}
            </span>
          ))}
          {idea.tags.length === 0 && (
            <span className="text-xs text-gray-400">No tags yet</span>
          )}
        </div>
        <form onSubmit={addTag} className="flex gap-2">
          <input
            value={newTag}
            onChange={e => setNewTag(e.target.value)}
            placeholder="Add tag..."
            className="flex-1 text-sm border border-gray-200 rounded-lg px-3 py-1.5 focus:outline-none focus:ring-2 focus:ring-indigo-200"
          />
          <button
            type="submit"
            className="text-sm bg-gray-100 hover:bg-gray-200 px-3 py-1.5 rounded-lg text-gray-700 transition-colors"
          >
            Add
          </button>
        </form>
      </div>

      {/* Actions */}
      <div className="flex gap-2 flex-wrap">
        {idea.status !== 'saved' && (
          <button
            onClick={() => updateStatus('saved')}
            disabled={saving}
            className="flex-1 bg-indigo-600 text-white py-2 rounded-lg text-sm font-medium hover:bg-indigo-700 disabled:opacity-50 transition-colors"
          >
            Save Idea
          </button>
        )}
        {idea.status !== 'inbox' && (
          <button
            onClick={() => updateStatus('inbox')}
            disabled={saving}
            className="flex-1 border border-gray-200 text-gray-600 py-2 rounded-lg text-sm hover:bg-gray-50 disabled:opacity-50 transition-colors"
          >
            Back to Inbox
          </button>
        )}
        {idea.status !== 'dismissed' && (
          <button
            onClick={() => updateStatus('dismissed')}
            disabled={saving}
            className="flex-1 border border-gray-200 text-gray-400 py-2 rounded-lg text-sm hover:bg-gray-50 disabled:opacity-50 transition-colors"
          >
            Dismiss
          </button>
        )}
        {idea.upwork_url && (
          <a
            href={idea.upwork_url}
            target="_blank"
            rel="noopener noreferrer"
            className="flex-1 border border-gray-200 text-gray-600 py-2 rounded-lg text-sm text-center hover:bg-gray-50 transition-colors"
          >
            View on Upwork ↗
          </a>
        )}
      </div>
    </div>
  );
}
```
